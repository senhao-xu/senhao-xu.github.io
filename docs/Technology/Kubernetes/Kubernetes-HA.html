环境初始化 ​

bash
#!/bin/bash
#设置颜色

log(){
 echo -e '\e[92m'$1'\e[0m'
}

log "start install kubernetes"

#系统配置
log "stop firewalld start"
systemctl disable firewalld --now
log "stop firewalld end"

log "stop selinux start"
setenforce 0
sed -i 's/enforcing/disabled/' /etc/selinux/config
log "stop selinux end"

log "stop swap start"
swapoff -a
sed -i 's/.*swap.*/#&/' /etc/fstab
log "stop swap end"

# 网络配置
log "update bridge start"
cat <<EOF >  /etc/sysctl.d/kubernetes.conf
net.bridge.bridge-nf-call-ip6tables = 1
net.bridge.bridge-nf-call-iptables = 1
net.ipv4.ip_forward=1
EOF
sysctl --system
modprobe br_netfilter
lsmod | grep br_netfilter
log "update bridge end"

log "config ipvs start"
rpm -Uvh --force --nodeps rpm/ipvs-rpm/*.rpm
cat > /etc/sysconfig/modules/ipvs.modules <<EOF
modprobe -- ip_vs
modprobe -- ip_vs_rr
modprobe -- ip_vs_wrr
modprobe -- ip_vs_sh
modprobe -- nf_conntrack_ipv4  
EOF
chmod +x /etc/sysconfig/modules/ipvs.modules 
# 执行脚本
/etc/sysconfig/modules/ipvs.modules
# 验证ipvs模块
lsmod | grep -e ip_vs -e nf_conntrack_ipv4
log "config ipvs end"

log "rpm install start"
rpm -Uvh --force --nodeps rpm/expect-rpm/*.rpm
log "rpm install end"

log "docker install start"
if ! [ -x "$(command -v docker)" ]; then
rpm -Uvh --force --nodeps rpm/docker-rpm/*.rpm
tee /etc/docker/daemon.json <<-'EOF'
{
  "registry-mirrors": [],
  "log-driver": "json-file",
  "log-opts": {
    "max-size": "100m",
    "max-file":"3"
  },
  "exec-opts": ["native.cgroupdriver=systemd"]
}
EOF
systemctl daemon-reload && systemctl restart docker && systemctl enable docker
fi
log "docker install end"

log "cri-docker install start"
rpm -Uvh --force --nodeps rpm/cri-dockerd/*.rpm
systemctl enable cri-docker --now
log "cri-docker install end"

log "docker images recovery start"
docker load -i images/kubernetes-1.28.2.tar
docker load -i images/kube-flannel-img.tar
docker load -i images/metrics-server.tar
log "docker images recovery end"

log "kubeadm install start"
if ! [ -x "$(command -v kubectl)" ]; then
rpm -Uvh --force --nodeps rpm/kubeadm-rpm/*.rpm
systemctl enable --now kubelet
fi
log "kubeadm install end"

ip=`ip route get 8.8.8.8 | awk 'NR==1{print $7}'`

log "kubernetes master start"
cat >> /etc/hosts << EOF
$ip master
EOF

log "kubernetes master end"

kube-vip ​

bash
export VIP=192.168.88.100
export INTERFACE=ens32
export KVVERSION=v1.0.0
docker run --rm --network host \
  ghcr.io/kube-vip/kube-vip:$KVVERSION \
  manifest pod \
  --interface $INTERFACE \
  --address $VIP \
  --controlplane \
  --services \
  --arp \
  --leaderElection | tee /etc/kubernetes/manifests/kube-vip.yaml
